-- RB4. Red Bull lines on S624100ZT0, Jan-Jun 2025: are they in pa_invoice, and with how much HL?
WITH g AS (
    SELECT DISTINCT refdocnr AS factura, regexp_replace(matnr, '^0+', '') AS material,
           date_format(budat, 'yyyy-MM') AS mes, fkart
    FROM brewdat_uc_delta_maz_prod.slv_maz_finance_sap_pr3.copecac_glpca
    WHERE __partition_column BETWEEN '202501' AND '202506'
      AND rbukrs = 'DC00' AND racct = 'S624100ZT0'
      AND budat BETWEEN '2025-01-01' AND '2025-06-30'
),
i AS (
    SELECT regexp_replace(trim(CAST(invoice_number AS STRING)), '^0+', '') AS factura,
           regexp_replace(trim(CAST(product_short_code AS STRING)), '^0+', '') AS material,
           COUNT(*) AS lineas, SUM(total_volume) AS hl
    FROM brewdat_uc_delta_maz_prod.gld_maz_sales_sell_in.pa_invoice
    WHERE invoice_date BETWEEN '2024-12-01' AND '2025-07-31'
      AND (op_ind IS NULL OR op_ind NOT IN ('D', 'DELETE'))
    GROUP BY 1, 2
)
SELECT g.mes, g.fkart AS tipo_factura,
       COUNT(*) AS lineas_cuenta,
       SUM(CASE WHEN i.factura IS NOT NULL THEN 1 ELSE 0 END) AS estan_en_pa_invoice,
       ROUND(SUM(COALESCE(i.hl, 0)), 2) AS hl_en_pa_invoice
FROM g LEFT JOIN i ON i.factura = g.factura AND i.material = g.material
GROUP BY 1, 2
ORDER BY 1, 2;
