let
    Sitio = "https://anheuserbuschinbev.sharepoint.com/sites/pa/fi",
    Carpeta = "https://anheuserbuschinbev.sharepoint.com/sites/pa/fi/IC/sql_server_historicos/deal_conditions/",
    Origen = SharePoint.Files(Sitio, [ApiVersion = 15]),
    Archivos = Table.SelectRows(Origen, each Text.Lower([Folder Path]) = Text.Lower(Carpeta) and Text.EndsWith(Text.Lower([Name]), ".csv")),
    ConTamano = Table.AddColumn(Archivos, "Tamano_MB", each Number.Round(Record.FieldOrDefault([Attributes], "Size", 0) / 1048576, 1), type number),
    ConLineas = Table.AddColumn(ConTamano, "Primeras_5_lineas", each Text.Combine(List.FirstN(Lines.FromBinary([Content], null, null, 65001), 5), "#(lf)"), type text),
    Resultado = Table.SelectColumns(ConLineas, {"Name", "Date modified", "Tamano_MB", "Primeras_5_lineas"})
in
    Resultado
