let
    Sitio = "https://anheuserbuschinbev.sharepoint.com/sites/pa/fi",
    Carpeta = "https://anheuserbuschinbev.sharepoint.com/sites/pa/fi/IC/sql_server_historicos/deal_conditions/",
    Origen = SharePoint.Files(Sitio, [ApiVersion = 15]),
    Archivos = Table.SelectRows(Origen, each Text.Lower([Folder Path]) = Text.Lower(Carpeta) and Text.EndsWith(Text.Lower([Name]), ".csv")),
    Leer = (contenido as binary) as table =>
        let
            Csv = Csv.Document(contenido, [Delimiter = ",", Encoding = 1252, QuoteStyle = QuoteStyle.Csv]),
            Enc = Table.PromoteHeaders(Csv, [PromoteAllScalars = true]),
            Sel = Table.SelectColumns(Enc, {"Calendar Day", "Sales Volume (HL) - BIL", "valor_condicion", "Denominación"}),
            Tip = Table.TransformColumns(Sel, {
                {"Calendar Day", each try Date.FromText(_, [Format = "yyyy-MM-dd", Culture = "en-US"]) otherwise null, type date},
                {"Sales Volume (HL) - BIL", each try Number.FromText(_, "en-US") otherwise null, type number},
                {"valor_condicion", each try Number.FromText(_, "en-US") otherwise null, type number}})
        in
            Tip,
    ConDatos = Table.AddColumn(Archivos, "Datos", each Leer([Content])),
    Solo = Table.SelectColumns(ConDatos, {"Name", "Datos"}),
    Exp = Table.ExpandTableColumn(Solo, "Datos", {"Calendar Day", "Sales Volume (HL) - BIL", "valor_condicion", "Denominación"}),
    ConMes = Table.AddColumn(Exp, "Mes", each if [Calendar Day] = null then null else Date.StartOfMonth([Calendar Day]), type date),
    Resumen = Table.Group(ConMes, {"Name", "Mes"}, {
        {"filas", each Table.RowCount(_), Int64.Type},
        {"primer_dia", each List.Min([Calendar Day]), type date},
        {"ultimo_dia", each List.Max([Calendar Day]), type date},
        {"hl", each Number.Round(List.Sum([#"Sales Volume (HL) - BIL"]), 2), type number},
        {"valor_condicion", each Number.Round(List.Sum([valor_condicion]), 2), type number},
        {"ejemplo_denominacion", each List.First(List.RemoveNulls([Denominación])), type text}}),
    Orden = Table.Sort(Resumen, {{"Mes", Order.Ascending}})
in
    Orden






SELECT LEFT(CONVERT(varchar(10), [Calendar Day], 23), 7) AS mes,
       COUNT(*) AS filas,
       ROUND(SUM(CAST([Sales Volume (HL) - BIL] AS float)), 2) AS hl,
       ROUND(SUM(CAST([valor_condicion] AS float)), 2) AS valor_condicion
FROM WarehousePanama.dbo.deal_conditions_fact
GROUP BY LEFT(CONVERT(varchar(10), [Calendar Day], 23), 7)
ORDER BY 1;
