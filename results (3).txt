-- =====================================================================================
-- A. REGALIAS NUEVAS (ZNO3) DESDE LAS FACTURAS (VBRK + VBRP), SIN GLPCA
--    Concilia calcula el volumen por clase de factura. Para llegar a los mismos 656.53 HL del 7 al 21-nov
--    (solo las lineas con ITBMS) hace falta saber que lineas tienen impuesto.
-- =====================================================================================

-- A1. Columnas de VBRP (para ver como se llama el campo del impuesto; en SAP es MWSBP)
DESCRIBE TABLE brewdat_uc_delta_maz_prod.slv_maz_sales_sap_pr3.copecac_vbrp;

-- A2. Facturas ZNO3 del 7 al 21-nov: volumen total y volumen de las lineas con impuesto
--     Esperado: hl_total = 830.32 y hl_con_impuesto = 656.53. Si falla por la columna mwsbp, mandarme el error y el resultado de A1.
WITH k AS (
    SELECT vbeln, fkart, CAST(fkdat AS DATE) AS dia, op_ind,
           ROW_NUMBER() OVER (PARTITION BY vbeln ORDER BY source_commit_ts DESC) AS rn
    FROM brewdat_uc_delta_maz_prod.slv_maz_sales_sap_pr3.copecac_vbrk
    WHERE __partition_column BETWEEN '202610' AND '202612'
      AND fkdat BETWEEN '2026-11-07' AND '2026-11-21' AND vkorg = 'DC01' AND TRIM(fkart) = 'ZNO3'
),
kv AS (
    SELECT vbeln, dia FROM k WHERE rn = 1 AND UPPER(TRIM(COALESCE(op_ind, ''))) NOT IN ('D', 'DELETE')
),
p AS (
    SELECT vbeln, posnr, matnr, volum, vkaus, mwsbp, op_ind,
           ROW_NUMBER() OVER (PARTITION BY vbeln, posnr ORDER BY source_commit_ts DESC) AS rn
    FROM brewdat_uc_delta_maz_prod.slv_maz_sales_sap_pr3.copecac_vbrp
    WHERE __partition_column BETWEEN '20261015' AND '20261130'
      AND vbeln IN (SELECT vbeln FROM kv)
)
SELECT TRIM(COALESCE(p.vkaus, '')) AS uso,
       COUNT(DISTINCT p.vbeln) AS facturas, COUNT(*) AS lineas,
       ROUND(SUM(p.volum) / 100000, 2) AS hl_total,
       ROUND(SUM(CASE WHEN COALESCE(p.mwsbp, 0) <> 0 THEN p.volum ELSE 0 END) / 100000, 2) AS hl_con_impuesto,
       SUM(CASE WHEN COALESCE(p.mwsbp, 0) <> 0 THEN 1 ELSE 0 END) AS lineas_con_impuesto
FROM kv JOIN p ON p.vbeln = kv.vbeln
WHERE p.rn = 1 AND UPPER(TRIM(COALESCE(p.op_ind, ''))) NOT IN ('D', 'DELETE')
GROUP BY 1
ORDER BY 1;
