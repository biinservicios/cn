-- VBAK: which date does the partition follow, and what is in the null partition?
SELECT COALESCE(__partition_column, '(null)') AS particion,
       COUNT(*) AS filas,
       MIN(erdat) AS erdat_min, MAX(erdat) AS erdat_max,
       MIN(audat) AS audat_min, MAX(audat) AS audat_max
FROM brewdat_uc_delta_maz_prod.slv_maz_sales_sap_pr3.copecac_vbak
WHERE __partition_column IN ('202610', '202611') OR __partition_column IS NULL
GROUP BY 1
ORDER BY 1;
