WITH g AS (
    SELECT LEFT(CONVERT(varchar(10), [Calendar Day], 23), 7) AS mes,
           [Factura], [Material],
           COUNT(*) AS filas,
           COUNT(DISTINCT ISNULL(CAST([promo] AS varchar(50)), '')) AS promos,
           SUM(CAST([Sales Volume (HL) - BIL] AS float)) AS hl_suma,
           MAX(CAST([Sales Volume (HL) - BIL] AS float)) AS hl_max,
           MIN(CAST([Sales Volume (HL) - BIL] AS float)) AS hl_min
    FROM WarehousePanama.dbo.deal_conditions_fact
    GROUP BY LEFT(CONVERT(varchar(10), [Calendar Day], 23), 7), [Factura], [Material]
)
SELECT mes,
       COUNT(*) AS lineas,
       SUM(filas) AS filas,
       SUM(CASE WHEN promos > 1 THEN 1 ELSE 0 END) AS lineas_con_varias_promos,
       SUM(CASE WHEN filas > promos THEN 1 ELSE 0 END) AS lineas_con_filas_repetidas,
       SUM(CASE WHEN hl_min <> hl_max THEN 1 ELSE 0 END) AS lineas_con_hl_distinto,
       ROUND(SUM(hl_suma), 2) AS hl_legado,
       ROUND(SUM(hl_max), 2) AS hl_una_fila_por_linea
FROM g
GROUP BY mes
ORDER BY mes;



SELECT date_format(invoice_date, 'yyyy-MM') AS mes,
       ROUND(SUM(total_volume), 2) AS hl_pa_invoice
FROM brewdat_uc_delta_maz_prod.gld_maz_sales_sell_in.pa_invoice
WHERE invoice_date >= '2024-06-01' AND invoice_date < '2026-09-01'
  AND (op_ind IS NULL OR op_ind NOT IN ('D', 'DELETE'))
GROUP BY 1
ORDER BY 1;




WITH ped AS (
    SELECT DISTINCT TRIM(aubel) AS aubel
    FROM brewdat_uc_delta_maz_prod.slv_maz_sales_sap_pr3.copecac_vbrp
    WHERE __partition_column >= date_format(date_sub(current_date(), 7), 'yyyyMMdd')
      AND erdat >= date_sub(current_date(), 7)
)
SELECT 'fbam' AS tabla, __partition_column AS particion, COUNT(*) AS filas
FROM brewdat_uc_delta_maz_prod.slv_maz_sales_sap_pr3.copecac_dsdpr_fbam
WHERE TRIM(vbeln_int) IN (SELECT aubel FROM ped)
GROUP BY __partition_column
UNION ALL
SELECT 'fbfg', __partition_column, COUNT(*)
FROM brewdat_uc_delta_maz_prod.slv_maz_sales_sap_pr3.copecac_dsdpr_fbfg
WHERE TRIM(vbeln_int) IN (SELECT aubel FROM ped)
GROUP BY __partition_column
ORDER BY 1, 2;
