-- D5. pa_invoice HL by invoice day, this month
SELECT invoice_date, ROUND(SUM(total_volume), 2) AS hl, COUNT(*) AS lineas
FROM brewdat_uc_delta_maz_prod.gld_maz_sales_sell_in.pa_invoice
WHERE invoice_date >= trunc(current_date(), 'MM')
  AND (op_ind IS NULL OR op_ind NOT IN ('D', 'DELETE'))
GROUP BY 1
ORDER BY 1;
